# Python List Methods - Complete Guide

# 1. append() - Add single element to end
# Modifies original list | Returns: None
lst = [1, 2, 3]
lst.append(4)
print(lst)  # [1, 2, 3, 4]

# 2. extend() - Add multiple elements from iterable
# Modifies original list | Returns: None
lst = [1, 2, 3]
lst.extend([4, 5, 6])
print(lst)  # [1, 2, 3, 4, 5, 6]

# 3. insert() - Insert element at specific index
# Modifies original list | Returns: None
lst = [1, 2, 4]
lst.insert(2, 3)
print(lst)  # [1, 2, 3, 4]

# 4. remove() - Remove first occurrence of value
# Modifies original list | Returns: None
lst = [1, 2, 3, 2]
lst.remove(2)
print(lst)  # [1, 3, 2]

# 5. pop() - Remove and return element at index (default: last)
# Modifies original list | Returns: removed element
lst = [1, 2, 3]
val = lst.pop(1)
print(val, lst)  # 2 [1, 3]

# 6. clear() - Remove all elements
# Modifies original list | Returns: None
lst = [1, 2, 3]
lst.clear()
print(lst)  # []

# 7. index() - Find index of first occurrence
# Does NOT modify list | Returns: index (int)
lst = [1, 2, 3, 2]
idx = lst.index(2)
print(idx)  # 1

# 8. count() - Count occurrences of value
# Does NOT modify list | Returns: count (int)
lst = [1, 2, 2, 3, 2]
cnt = lst.count(2)
print(cnt)  # 3

# 9. sort() - Sort list in place
# Modifies original list | Returns: None
lst = [3, 1, 2]
lst.sort()
print(lst)  # [1, 2, 3]

# 10. reverse() - Reverse list in place
# Modifies original list | Returns: None
lst = [1, 2, 3]
lst.reverse()
print(lst)  # [3, 2, 1]

# 11. copy() - Create shallow copy
# Does NOT modify list | Returns: new list
lst = [1, 2, 3]
lst2 = lst.copy()
print(lst2)  # [1, 2, 3]

# ===== BUILT-IN FUNCTIONS FOR LISTS =====

# len() - Get list length
# Does NOT modify | Returns: int
lst = [1, 2, 3, 4]
print(len(lst))  # 4

# max() - Find maximum value
# Does NOT modify | Returns: max element
lst = [1, 5, 3, 2]
print(max(lst))  # 5

# min() - Find minimum value
# Does NOT modify | Returns: min element
lst = [1, 5, 3, 2]
print(min(lst))  # 1

# sum() - Sum all elements
# Does NOT modify | Returns: sum (int/float)
lst = [1, 2, 3, 4]
print(sum(lst))  # 10

# sorted() - Return sorted copy
# Does NOT modify original | Returns: new sorted list
lst = [3, 1, 2]
new_lst = sorted(lst)
print(new_lst)  # [1, 2, 3]
print(lst)     # [3, 1, 2] (original unchanged)

# reversed() - Return reversed iterator
# Does NOT modify | Returns: iterator object
lst = [1, 2, 3]
rev = reversed(lst)
print(list(rev))  # [3, 2, 1]
print(lst)        # [1, 2, 3] (original unchanged)

# enumerate() - Get index and value pairs
# Does NOT modify | Returns: enumerate object
lst = ['a', 'b', 'c']
for idx, val in enumerate(lst):
  print(idx, val)  # 0 a | 1 b | 2 c

# any() - Check if any element is True
# Does NOT modify | Returns: bool
lst = [False, False, True]
print(any(lst))  # True

# all() - Check if all elements are True
# Does NOT modify | Returns: bool
lst = [True, True, True]
print(all(lst))  # True

# list() - Convert iterable to list
# Does NOT modify | Returns: new list
tup = (1, 2, 3)
new_lst = list(tup)
print(new_lst)  # [1, 2, 3]
