# Tuple Methods and Built-in Functions.


# ===== BUILT-IN METHODS FOR TUPLE =====

# Note: Tuples are immutable, so no methods like append(), remove(), etc.

# 1. count() - Returns number of occurrences of a value
# Return type: int
t = (1, 2, 2, 3, 2, 4)
print(t.count(2))  # Output: 3

# 2. index() - Returns index of first occurrence
# Does NOT modify list | Returns: index (int)
# Raises ValueError if not found
t = ('a', 'b', 'c', 'b')
print(t.index('b'))  # Output: 1


# ===== BUILT-IN FUNCTIONS FOR TUPLE =====


# 3. len() - Returns length of tuple
# Return type: int
t = (1, 2, 3)
print(len(t))  # Output: 3

# 4. max() - Returns maximum value
# Return type: same as tuple elements
t = (5, 2, 9, 1)
print(max(t))  # Output: 9

# 5. min() - Returns minimum value
# Return type: same as tuple elements
t = (5, 2, 9, 1)
print(min(t))  # Output: 1

# 6. sum() - Returns sum of numeric values
# Return type: int/float
t = (1, 2, 3, 4)
print(sum(t))  # Output: 10

# 7. sorted() - Returns sorted list from tuple
# Return type: list
t = (3, 1, 4, 1, 5)
print(sorted(t))  # Output: [1, 1, 3, 4, 5]

# 8. reversed() - Returns reverse iterator
# Return type: iterator
t = (1, 2, 3)
print(list(reversed(t)))  # Output: [3, 2, 1]

# 9. enumerate() - Returns (index, value) pairs
# Return type: enumerate object
t = ('a', 'b', 'c')
for i, val in enumerate(t):
  print(i, val)  # Output: 0 a, 1 b, 2 c

# 10. any() - Returns True if any element is truthy
# Return type: bool
t = (0, 0, 1)
print(any(t))  # Output: True

# 11. all() - Returns True if all elements are truthy
# Return type: bool
t = (1, 2, 3)
print(all(t))  # Output: True

# 12. zip() - Combines multiple tuples
# Return type: zip object
t1 = (1, 2, 3)
t2 = ('a', 'b', 'c')
print(list(zip(t1, t2)))  # Output: [(1, 'a'), (2, 'b'), (3, 'c')]

# 13. type() - Returns type of tuple
# Return type: type object
t = (1, 2, 3)
print(type(t))  # Output: <class 'tuple'>

# 14. isinstance() - Checks if object is tuple
# Return type: bool
t = (1, 2, 3)
print(isinstance(t, tuple))  # Output: True

# 15. tuple() - Converts iterable to tuple
# Return type: tuple
print(tuple([1, 2, 3]))  # Output: (1, 2, 3)
